import { TrendingUp, Check, RefreshCw } from "lucide-react";
import { Goal } from "@/shared/types";

interface GoalCardProps {
  goal: Goal;
  onComplete: (goalId: number) => void;
  isCompleted: boolean;
  isLoading: boolean;
}

export default function GoalCard({ goal, onComplete, isCompleted, isLoading }: GoalCardProps) {
  const Icon = TrendingUp;
  
  const handleToggleCompletion = () => {
    if (!isLoading) {
      onComplete(goal.id);
    }
  };

  return (
    <div 
      className={`p-4 rounded-xl border-2 transition-all duration-300 flex items-center justify-between ${
        isCompleted
          ? "border-pink-400 bg-pink-50 shadow-md"
          : "border-indigo-200 bg-white hover:border-indigo-400 hover:shadow-lg"
      }`}
    >
      <div className="flex items-center gap-4">
        <div className={`w-10 h-10 rounded-full flex items-center justify-center ${isCompleted ? 'bg-pink-500 text-white' : 'bg-indigo-100 text-indigo-600'}`}>
          <Icon className="w-5 h-5" />
        </div>
        <div>
          <h3 className="text-lg font-semibold text-gray-800 line-clamp-1">{goal.title}</h3>
          <p className="text-sm text-gray-500">
            Meta: {goal.target_frequency}x/semana
          </p>
        </div>
      </div>
      
      <button
        onClick={handleToggleCompletion}
        disabled={isLoading}
        className={`w-10 h-10 rounded-full flex items-center justify-center transition-all duration-300 ${
          isCompleted
            ? "bg-pink-500 text-white hover:bg-pink-600"
            : "bg-indigo-500 text-white hover:bg-indigo-600"
        } disabled:opacity-50`}
        title={isCompleted ? "Desmarcar meta de hoje" : "Marcar meta de hoje"}
      >
        {isLoading ? <RefreshCw className="w-5 h-5 animate-spin" /> : isCompleted ? <Check className="w-5 h-5" /> : <Check className="w-5 h-5" />}
      </button>
    </div>
  );
}
